import React, { useState } from 'react';
import Logout from './Logout';

function TodoApp({ onLogout, username }) { // Accept the onLogout prop
    const [todoInput, updateInput] = useState('');
    const [todoList, updateTodo] = useState([]);
    const [inputError, setInputError] = useState(false);
    const [showLogoutConfirmation, setShowLogoutConfirmation] = useState(false);

    let nextID = todoList.length > 0 ? todoList[todoList.length - 1].id + 1 : 1;

    function addNewTodo() {
        if (!todoInput || todoInput.trim() === "") {
            setInputError(true);
        } else {
            setInputError(false);
            const newTodo = [
                ...todoList,
                {
                    id: nextID,
                    task: todoInput
                }
            ];
            updateTodo(newTodo);
            updateInput("");
        }
    }

    function deleteTodo(id) {
        const updatedTodo = todoList.filter(todo => todo.id !== id);
        updateTodo(updatedTodo);
    }

    const handleLogout = () => {
        // Show confirmation popup
        const confirmLogout = window.confirm("Are you sure you want to logout?");
        if (confirmLogout) {
            onLogout(); // Call onLogout if confirmed
        }
    };

    const confirmLogout = () => {
        onLogout(); // Call onLogout to log out
        setShowLogoutConfirmation(false); // Hide confirmation dialog
    };

    return (
        <div className="container mt-5 w-50">
            <div className="d-flex justify-content-between align-items-center mb-3">
                <h2 className='centered-title'>Todo App Using React</h2>
                <div>
                    <span className="text-muted">Logged in as: ndiddi</span>
                </div>
            </div>
            
            <div className="input-group">
                <input
                    className="form-control"
                    onChange={(e) => {
                        let task = e.target.value;
                        updateInput(task);
                        setInputError(false);
                    }}
                    type='text'
                    value={todoInput}
                    placeholder='Please enter your task'
                />
                <button onClick={addNewTodo} className="btn btn-primary">Add</button>
            </div>

            {inputError && (<div className="text-danger mt-1">Please enter a valid task.</div>)}

            <ul className="list-group mt-4">
                {todoList.slice().reverse().map((todo) => (
                    <li key={todo.id} className="list-group-item">
                        <p>{todo.task}</p>
                        <button onClick={() => deleteTodo(todo.id)} className="btn">❌</button>
                    </li>
                ))}
            </ul>

            {/* Logout Button */}
            <div className='button-center'>
            <button onClick={handleLogout} className="btn btn-danger mt-3">Logout</button>
            </div>

            {showLogoutConfirmation && (
                <Logout onLogout={confirmLogout} />
            )}
        </div>
    );
}

export default TodoApp;
